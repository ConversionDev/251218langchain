# EC2에서 아래 내용으로 교체할 server 블록 (api.kanggyeonggu.store, 443)
# 기존 "server { ... server_name api.kanggyeonggu.store; ... location / { try_files ... } ... }" 전체를
# 이 블록으로 교체하고, 파일 맨 위/맨 아래에 있는 server 밖의 location / 조각은 삭제하세요.

server {
        root /var/www/html;
        index index.html index.htm index.nginx-debian.html;
        server_name api.kanggyeonggu.store; # managed by Certbot

        location / {
                proxy_pass http://127.0.0.1:8000;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
        }

        listen [::]:443 ssl ipv6only=on; # managed by Certbot
        listen 443 ssl; # managed by Certbot
        ssl_certificate /etc/letsencrypt/live/api.kanggyeonggu.store/fullchain.pem; # managed by Certbot
        ssl_certificate_key /etc/letsencrypt/live/api.kanggyeonggu.store/privkey.pem; # managed by Certbot
        include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
        ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}
